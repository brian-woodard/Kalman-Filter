typedef union
{
    long    l;
    float   f;
} LfUnion;

float doBSCALE(int width, int scale)
{
    LfUnion lf;

    lf.l = (127 + (width) - (scale)) << 23;

    return lf.f;
}

double HALF_CIRCLE_DEGREES = 180.0;
double revsToDegs32(int32_t value)
{
    return static_cast<double>(value * HALF_CIRCLE_DEGREES / doBSCALE(31, 0));
}
